function post_like(id) {
    const userId = <?= $_SESSION['user_id']; ?>; // Include this dynamically from PHP
    const index = selectedIcons.indexOf(id);

    const data = {
        id: id,
        user_id: userId, // Send the user ID
        likes: index === -1
    };

    $.ajax({
        type: "POST",
        url: '<?= ROOT ?>/CWDramaLike',
        data: data,
        cache: false,
        success: function (res) {
            console.log(res);
        },
        error: function (xhr, status, error) {
            console.error(error);
        },
    });

    // Update local state and UI
    if (index === -1) {
        selectedIcons.push(id);
        document.querySelector(`.icon${id}`).classList.add('selected');
    } else {
        selectedIcons.splice(index, 1);
        document.querySelector(`.icon${id}`).classList.remove('selected');
    }

    localStorage.setItem('selectedIcons', JSON.stringify(selectedIcons));
}

    function limitWords($text, $limit)
    {
        $words = explode(" ", $text);
        $limitedWords = array_slice($words, 0, $limit);
        $result = implode(" ", $limitedWords);

        if (count($words) > $limit) {
            $result .= '...';
        }

        return $result;

            const icon = document.querySelector(`#iconHeart${articleId}`);
            const isLiked = icon.classList.contains('selected'); // Check if the article is already liked

            // Toggle the like state visually
            if (isLiked) {
                icon.classList.remove('selected');
            } else {
                icon.classList.add('selected');
            }

            // Send the like or unlike request to the server
            const data = {
                id: articleId,
                user_id: currentUser,
                likes: !isLiked // true if liking, false if unliking
            };

            fetch('<?= ROOT ?>/CWDramaLike', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json'
                },
                body: JSON.stringify(data)
            })
                .then(response => response.json())
                .then(json => {
                    document.querySelector(`#likeCount${articleId}`).textContent = json.likes;
                })
                .catch(error => console.error('Error:', error));

class CWDramaLike
{
    use Controller;

    public function index()
    {
        $article = new Article();

        // Retrieve POST data
        $articleId = $_POST['id'];
        $userId = $_POST['user_id'];
        $isLiked = filter_var($_POST['likes'], FILTER_VALIDATE_BOOLEAN); // Convert "true"/"false" to boolean

        // Fetch the article
        $result = $article->first(['id' => $articleId]);

        if (!$result) {
            echo json_encode(['error' => 'Article not found']);
            return;
        }

        $likedBy = json_decode($result->liked_by, true) ?? []; // Decode liked_by column (JSON) to an array

        if ($isLiked) {
            // Add the user ID if not already in the liked_by list
            if (!in_array($userId, $likedBy)) {
                $likedBy[] = $userId;
                $data['likes'] = $result->likes + 1; // Increment likes count
            }
        } else {
            // Remove the user ID if it's in the liked_by list
            if (($key = array_search($userId, $likedBy)) !== false) {
                unset($likedBy[$key]);
                $data['likes'] = $result->likes - 1; // Decrement likes count
            }
        }

        $data['liked_by'] = json_encode(array_values($likedBy)); // Re-encode liked_by as JSON

        // Update the article
        $article->update($articleId, $data);

        // Return the updated likes count
        echo json_encode(['likes' => $data['likes']]);
    }
}
